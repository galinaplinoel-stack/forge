def binary_search(arr, target, find_first=True):
"""Binary search with duplicate handling.
Args:
arr: Sorted list of elements
target: Element to search for
find_first: If True, find first occurrence
Returns:
Index of target, or -1 if not found
"""
left, right = 0, len(arr) - 1
result = -1
while left <= right:
mid = (left + right) // 2
if arr[mid] == target:
result = mid
if find_first:
right = mid - 1
else:
left = mid + 1
elif arr[mid] < target:
left = mid + 1
else:
right = mid - 1
return result
